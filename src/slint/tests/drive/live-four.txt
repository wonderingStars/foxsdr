#! frames=900 size=1600x900
# live-four: four LIVE radios in a 2 x 2 grid (screen stage S4, job B; docs/multi-radio-records/screen-s4.md). Radio 1 is the engine's own; radios 2 to 4 are GENERATOR radios opened through the
# engine's real RADIO_ADD (`radio add <n> <tone Hz>`), each with a tone of its own: 50, 80 and 110 kHz above its centre. Each pane draws its OWN radio: the highest bin of each pane's last frame is
# at that radio's tone (state workspace.peakoff.<n>); a click in pane 3's spectrum retunes radio 3 only; STOP in pane 4 stops radio 4 only; removing radio 3 empties its cell and disturbs no other
# pane. Tick rows 4200-4216 on live radios (the S3 rows were shown with made-up ones).
10 state workspace.radios == 1
+1 radio add 2 50000
+1 radio add 3 80000
+1 radio add 4 110000
+30 state workspace.radios == 4
+1 click "2 x 2"
+15 state workspace.shown == 2
state workspace.cells == "1,2,3,4"
+20 state workspace.ticks.4 > 3
state workspace.peakoff.1 > 290000
state workspace.peakoff.1 < 310000
state workspace.peakoff.2 > 45000
state workspace.peakoff.2 < 55000
state workspace.peakoff.3 > 75000
state workspace.peakoff.3 < 85000
state workspace.peakoff.4 > 105000
state workspace.peakoff.4 < 115000
+1 state workspace.r.1.running == 1
state workspace.r.2.running == 1
state workspace.r.3.running == 1
state workspace.r.4.running == 1
+1 shot "live-four.png"
# tuning in pane 3 moves radio 3 only
+1 state workspace.r.3.tuned == 100300000
state workspace.r.2.tuned == 100300000
+1 press "Radio 3 spectrum"
+15 state workspace.r.3.tuned != 100300000
state workspace.r.1.tuned == 100300000
state workspace.r.2.tuned == 100300000
state workspace.r.4.tuned == 100300000
state workspace.selected == 1
# STOP in pane 4 stops radio 4 only
+1 click "Stop Radio 4"
+15 state workspace.r.4.running == 0
state workspace.r.1.running == 1
state workspace.r.2.running == 1
state workspace.r.3.running == 1
state workspace.selected == 1
+1 expect "Start Radio 4" enabled equals "true"
+1 click "Start Radio 4"
+15 state workspace.r.4.running == 1
# removing radio 3 empties its cell and disturbs no other pane
+1 radio remove 3
+25 state workspace.radios == 3
state workspace.cells == "1,2,0,4"
+1 state workspace.r.2.running == 1
state workspace.r.4.running == 1
state workspace.peakoff.2 > 45000
state workspace.peakoff.2 < 55000
state workspace.peakoff.4 > 105000
state workspace.peakoff.4 < 115000
+1 shot "live-four-removed-3.png"
