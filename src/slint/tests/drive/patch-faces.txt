# Every figure is read back from the pictures the faces were painted from (`state patchface.s<k>.<what>`, k the part's number among the Spectrum parts in the draft's order), and the
# window is captured so the pixels can be measured as well (shot).
10 state count.patchNodes == 6
+1 click "Fit"
# BEFORE THE PATCH RUNS: the window's two sentences, no picture, no figure from a frame that never came
+3 state patchface.s0.live == 0
state patchface.s0.sentence contains "Wire the radio or a channel here to see its spectrum."
state patchface.s1.live == 0
state patchface.s1.sentence contains "No spectrum yet - its radio is not running."
state patchface.s2.live == 0
state patchface.s2.sentence contains "Wire the radio or a channel here to see its spectrum."
state patchface.s0.frames == 0
# RUN IT
+1 click "START"
+40 state patchRunning == 1
+60 state patchface.s0.live == 1
state patchface.s1.live == 1
state patchface.s2.live == 0
state patchface.s0.sentence == ""
state patchface.s1.sentence == ""
state patchface.s2.sentence contains "Wire the radio or a channel here to see its spectrum."

state patchface.s0.span == 2000000
state patchface.s0.channel == 0
state patchface.s0.peakfrac > 0.63
state patchface.s0.peakfrac < 0.67
state patchface.s0.abovefloor > 40
state patchface.s0.markers == 1
state patchface.s0.markernames contains Channel
# THE CHANNEL-FED PART SHOWS ITS SLICE: narrower than the capture by more than ten times, the same tone in the middle of it
state patchface.s1.channel == 1
state patchface.s1.span < 200000
state patchface.s1.span > 1000
state patchface.s1.peakfrac > 0.40
state patchface.s1.peakfrac < 0.60
state patchface.s1.abovefloor > 20
state patchface.s1.markers == 1
# A MOVING WATERFALL: rows keep arriving and the picture changes between two looks a second and a half apart (the first look sets the reference)
state patchface.s0.rows >= 2
state patchface.s0.moved == 1
+45 state patchface.s0.moved == 1
state patchface.s1.moved == 1
+45 state patchface.s1.moved == 1
+1 shot "pf-green-1.png"

+40 state patchface.s0.framesdelta == 0

+30 state patchface.s1.framesdelta == 0
